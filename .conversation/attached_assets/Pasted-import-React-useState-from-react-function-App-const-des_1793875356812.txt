import React, { useState } from 'react';

function App() {
  const [description, setDescription] = useState('');
  const [tasks, setTasks] = useState([]);
  const [loading, setLoading] = useState(false);

  // دالة محاكاة توليد المهام بالذكاء الاصطناعي (يمكن ربطها بالباك إند لاحقاً)
  const handleGenerateTasks = () => {
    setLoading(true);
    setTimeout(() => {
      setTasks([
        { id: 1, title: "تحليل المتطلبات وقراءة الواجب", status: "To Do", assignee: "أحمد" },
        { id: 2, title: "تصميم قاعدة البيانات والواجهات", status: "In Progress", assignee: "سارة" },
        { id: 3, title: "برمجة النظام والربط", status: "To Do", assignee: "جود" }
      ]);
      setLoading(false);
    }, 1000);
  };

  return (
    <div className="min-h-screen bg-gray-50 p-6 font-sans" dir="rtl">
      <div className="max-w-4xl mx-auto">
        <header className="mb-8 text-center">
          <h1 className="text-3xl font-bold text-gray-800">منصة تنظيم مشاريع المجموعات</h1>
          <p className="text-gray-500 mt-2">حوّل وصف مشروعك الجامعي إلى مهام منظمة وموزعة بعدالة بضغطة زر</p>
        </header>

        {/* صندوق إدخال وصف المشروع */}
        <div className="bg-white p-6 rounded-xl shadow-md mb-8">
          <label className="block text-lg font-medium text-gray-700 mb-2">ألصق وصف المشروع أو الـ Syllabus هنا:</label>
          <textarea 
            className="w-full h-32 p-4 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:outline-none mb-4"
            placeholder="مثال: مطلوب بناء تطبيق لإدارة العيادات الطبية باستخدام Node.js و React..."
            value={description}
            onChange={(e) => setDescription(e.target.value)}
          />
          <button 
            onClick={handleGenerateTasks}
            className="bg-blue-600 hover:bg-blue-700 text-white font-medium px-6 py-3 rounded-lg transition duration-200 w-full md:w-auto"
          >
            {loading ? "جاري التحليل بالذكاء الاصطناعي..." : "توليد المهام الذكية 🚀"}
          </button>
        </div>

        {/* لوحة المهام (Dashboard) */}
        {tasks.length > 0 && (
          <div className="bg-white p-6 rounded-xl shadow-md">
            <h2 className="text-xl font-bold text-gray-800 mb-4">مهام المشروع المقترحة</h2>
            <div className="grid gap-4">
              {tasks.map((task) => (
                <div key={task.id} className="flex items-center justify-between p-4 border border-gray-200 rounded-lg bg-gray-50">
                  <div>
                    <h3 className="font-semibold text-gray-800">{task.title}</h3>
                    <span className="text-sm text-gray-500">المسؤول: {task.assignee}</span>
                  </div>
                  <span className={`px-3 py-1 rounded-full text-xs font-medium ${
                    task.status === 'In Progress' ? 'bg-yellow-100 text-yellow-800' : 'bg-blue-100 text-blue-800'
                  }`}>
                    {task.status}
                  </span>
                </div>
              ))}
            </div>
          </div>
        )}
      </div>
    </div>
  );
}

export default App;
